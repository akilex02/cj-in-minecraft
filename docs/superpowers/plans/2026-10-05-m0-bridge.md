# CJ en Minecraft — M0: el puente — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Que el jugador de Minecraft se mueva y CJ aparezca en GTA SA en las coordenadas equivalentes, con GTA ejecutándose sin foco, y comprobar que el exe de 14 MB y las direcciones de `gta-reversed` sirven.

**Architecture:** Dos procesos hablando UDP por `127.0.0.1`. Un plugin `.asi` de 32 bits (`gta-bridge`) parchea la llamada a `CGame::Process`, lee los paquetes de estado y teletransporta a CJ (sin gravedad ni colisión) cada frame, devolviendo un eco con su posición real. Un mod Fabric (`cj-fabric`) manda el estado del jugador a 60 Hz y muestra el eco en la barra de acción. Un cliente Python falso hace de oráculo para probar el puente sin Minecraft.

**Tech Stack:** C++20 cruzado a Windows 32 bits con `mingw-w64-gcc` (se prueba en Wine 11), CMake + Ninja; Python 3 (`unittest`); Java 21, Gradle y Fabric Loom para Minecraft 1.21.1 (yarn `1.21.1+build.3`, loader `0.19.5`, fabric-api `0.116.17+1.21.1`).

**Spec:** `docs/superpowers/specs/2026-10-05-cj-in-minecraft-design.md` (secciones 3, 4 y 7-M0).

**Alcance de este plan:** solo M0. M1 (CJ visible), M2 (combate), M3 (mundo) y M4 (pulido) tendrán cada uno su propio plan, escrito cuando el anterior esté verificado.

**Sin git:** el proyecto no es un repositorio git (el usuario decidió no hacer `git init` todavía). Por eso no hay pasos de commit; cada tarea termina con su verificación.

## Global Constraints

- Un solo jugador y sin conexión; todo el tráfico va por `127.0.0.1`.
- Los assets de CJ salen de la instalación de GTA SA del usuario y no se copian al repositorio (M0 no usa ningún asset).
- Minecraft **1.21.1 con Fabric**, Java 21, en un proyecto nuevo (no se tocan los mundos ni las instancias existentes de Prism).
- GTA SA: `GTA San Andreas/gta_sa.exe` de 14,383,616 bytes. El plugin es un `.asi` de 32 bits que **no sustituye funciones del juego**; `gta-reversed-master/` se usa solo como mapa de direcciones y estructuras.
- Protocolo: UDP en 127.0.0.1, binario, little-endian, con `magic` y `versión` en cada paquete. Minecraft → GTA a 60 Hz.
- GTA no debe pausarse sin foco.
- Si GTA deja de recibir estado, CJ queda en reposo (el plugin le devuelve gravedad y colisión).
- Ejes: GTA es Z arriba y Minecraft Y arriba; escala 1:1.

```python
#!/usr/bin/env python3
"""Checks that gta_sa.exe matches the 1.0 US address map used by gta-reversed."""
import struct
import sys
from pathlib import Path

IMAGE_BASE = 0x400000


def load_sections(data):
    pe = struct.unpack_from("<I", data, 0x3C)[0]
    count = struct.unpack_from("<H", data, pe + 6)[0]
    opt_size = struct.unpack_from("<H", data, pe + 20)[0]
    table = pe + 24 + opt_size
    sections = []
    for i in range(count):
        _, vsize, va, rsize, raw = struct.unpack_from("<8sIIII", data, table + i * 40)
        sections.append((IMAGE_BASE + va, min(vsize, rsize), raw))
    return sections


def read_va(data, sections, va, size):
    for start, length, raw in sections:
        if start <= va and va + size <= start + length:
            offset = raw + va - start
            return data[offset:offset + size]
    raise ValueError(f"VA {va:#x} is not backed by the file")


def main():
    path = Path(sys.argv[1] if len(sys.argv) > 1 else "GTA San Andreas/gta_sa.exe")
    data = path.read_bytes()
    sections = load_sections(data)
    failures = 0

    def check(name, ok, detail=""):
        nonlocal failures
        print(("OK   " if ok else "FAIL ") + name + (f" ({detail})" if detail else ""))
        failures += 0 if ok else 1

    # CWorld::FindPlayerPed: mov eax,[esp+4]; test eax,eax; jge +7; movzx eax,[PlayerInFocus 0xB7CD74];
    # imul eax,eax,0x190 (sizeof CPlayerInfo); mov eax,[eax+Players 0xB7CD98]
    expected = bytes.fromhex("8b44240485c07d070fb60574cdb70069c0900100008b8098cdb7")
    got = read_va(data, sections, 0x56E210, len(expected))
    check("FindPlayerPed @0x56E210 matches gta-reversed", got == expected, got.hex())

    # Idle() calls CGame::Process at 0x53E981 (E8 rel32 -> 0x53BEE0)
    call = read_va(data, sections, 0x53E981, 5)
    target = (0x53E981 + 5 + struct.unpack("<i", call[1:])[0]) & 0xFFFFFFFF
    check("call site @0x53E981 is E8 -> CGame::Process 0x53BEE0", call[0] == 0xE8 and target == 0x53BEE0, f"{call.hex()} -> {target:#x}")

    # CGame::Process prologue
    prologue = read_va(data, sections, 0x53BEE0, 7)
    check("CGame::Process prologue @0x53BEE0", prologue == bytes.fromhex("83ec0c535657e8"), prologue.hex())

    print("size:", len(data), "bytes")
    sys.exit(1 if failures else 0)


if __name__ == "__main__":
    main()
```

`protocol/cjproto.py`:

```python
"""Reference implementation of the CJ-in-Minecraft wire protocol (see PROTOCOL.md)."""
import math
import struct
from dataclasses import dataclass

MAGIC = 0x434A4D43
VERSION = 1
TYPE_STATE = 1
TYPE_ECHO = 2

FLAG_ON_GROUND = 1 << 0
FLAG_SPRINT = 1 << 1
FLAG_SNEAK = 1 << 2
FLAG_SWIM = 1 << 3
FLAG_CLIMBING = 1 << 4

INTENT_AIM = 1 << 0
INTENT_FIRE = 1 << 1
INTENT_MELEE = 1 << 2

_HEADER = struct.Struct("<IHH")
_STATE = struct.Struct("<IIffffffffIIHH")  # seq, ts, pos[3], vel[3], yaw, pitch, flags, intents, weapon, pad
_ECHO = struct.Struct("<IIfff")  # ack_seq, gta_time_ms, pos[3]
STATE_SIZE = _HEADER.size + _STATE.size  # 60
ECHO_SIZE = _HEADER.size + _ECHO.size  # 28


@dataclass(frozen=True)
class State:
    seq: int
    ts_ms: int
    pos: tuple
    vel: tuple
    yaw: float
    pitch: float
    flags: int = 0
    intents: int = 0
    weapon: int = 0


@dataclass(frozen=True)
class Echo:
    ack_seq: int
    gta_time_ms: int
    pos: tuple


def _check_header(data, expected_type, expected_size):
    if len(data) != expected_size:
        raise ValueError(f"expected {expected_size} bytes, got {len(data)}")
    magic, version, ptype = _HEADER.unpack_from(data, 0)
    if magic != MAGIC:
        raise ValueError("bad magic")
    if version != VERSION:
        raise ValueError(f"unsupported version {version}")
    if ptype != expected_type:
        raise ValueError(f"unexpected packet type {ptype}")


def pack_state(s):
    return _HEADER.pack(MAGIC, VERSION, TYPE_STATE) + _STATE.pack(
        s.seq, s.ts_ms, *s.pos, *s.vel, s.yaw, s.pitch, s.flags, s.intents, s.weapon, 0)


def unpack_state(data):
    _check_header(data, TYPE_STATE, STATE_SIZE)
    seq, ts, px, py, pz, vx, vy, vz, yaw, pitch, flags, intents, weapon, _pad = _STATE.unpack_from(data, _HEADER.size)
    return State(seq, ts, (px, py, pz), (vx, vy, vz), yaw, pitch, flags, intents, weapon)


def pack_echo(e):
    return _HEADER.pack(MAGIC, VERSION, TYPE_ECHO) + _ECHO.pack(e.ack_seq, e.gta_time_ms, *e.pos)


def unpack_echo(data):
    _check_header(data, TYPE_ECHO, ECHO_SIZE)
    ack, gta_time, x, y, z = _ECHO.unpack_from(data, _HEADER.size)
    return Echo(ack, gta_time, (x, y, z))


def mc_delta_to_gta(d):
    """Minecraft (x east, y up, z south) displacement -> GTA (x east, y north, z up)."""
    return (d[0], -d[2], d[1])


def mc_yaw_to_gta_heading(yaw_deg):
    """Minecraft yaw in degrees -> GTA heading in radians, in [0, 2*pi)."""
    return (math.pi - math.radians(yaw_deg)) % (2 * math.pi)
```

```markdown
# Protocolo CJ-Minecraft v1

UDP en `127.0.0.1`, binario, little-endian. Puertos: GTA escucha en **34560** (estado de Minecraft); Minecraft escucha en **34561** (eco de GTA). Solo un proceso puede ocupar el 34561 a la vez (mod o cliente falso).

Todos los paquetes empiezan con una cabecera de 8 bytes: `u32 magic = 0x434A4D43`, `u16 version = 1`, `u16 type`. Un paquete con tamaño, magic, versión o tipo incorrectos se ignora.

## State (type 1, Minecraft → GTA, 60 bytes, a 60 Hz)

| Offset | Tipo | Campo |
|---|---|---|
| 8 | u32 | seq (contador creciente; la comparación es con signo para tolerar el desbordamiento) |
| 12 | u32 | ts_ms (ms desde un origen arbitrario de Minecraft) |
| 16 | f32×3 | pos (coordenadas de Minecraft, x este / y arriba / z sur, en bloques) |
| 28 | f32×3 | vel (bloques/s) |
| 40 | f32 | yaw (grados de Minecraft) |
| 44 | f32 | pitch (grados de Minecraft) |
| 48 | u32 | flags: 1 en suelo, 2 sprint, 4 agachado, 8 nadando, 16 trepando |
| 52 | u32 | intents: 1 apuntar, 2 disparar, 4 golpear |
| 56 | u16 | weapon (id de arma, 0 = ninguna) |
| 58 | u16 | reservado (0) |

## Echo (type 2, GTA → Minecraft, 28 bytes, un paquete por frame de GTA mientras haya marioneta)

| Offset | Tipo | Campo |
|---|---|---|
| 8 | u32 | ack_seq (último `seq` aceptado) |
| 12 | u32 | gta_time_ms (`GetTickCount` de GTA) |
| 16 | f32×3 | pos real de CJ en GTA (x este / y norte / z arriba) |

## Ejes y anclaje

- GTA es Z arriba, Y norte; Minecraft es Y arriba, Z sur. Desplazamiento: `GTA = (dx, -dz, dy)`. Escala 1:1.
- Heading de GTA (rad) = `(π − yaw_mc_rad) mod 2π` (se usa desde M1).
- **Anclaje:** el primer `State` aceptado fija `mc_anchor = pos` y `gta_anchor = posición actual de CJ`. Desde entonces `target = gta_anchor + (pos − mc_anchor)` mapeado. El objetivo se recorta a x,y ∈ [−2900, 2900] y z ∈ [−100, 1500].
- Un `State` es válido si todos sus floats son finitos, |coordenada| ≤ 30 000 000 y su `seq` es posterior (con signo) al último aceptado.
- Si pasan más de 1000 ms sin `State` aceptado, la marioneta expira: se devuelve gravedad y colisión a CJ y se olvida el anclaje y la secuencia.
```

`gta-bridge/src/protocol.h`:

```cpp
#pragma once
#include <bit>
#include <cstddef>
#include <cstdint>
#include <cstring>

namespace cj {

static_assert(std::endian::native == std::endian::little, "the wire format is little-endian");

constexpr uint32_t kMagic = 0x434A4D43;
constexpr uint16_t kVersion = 1;
enum class PacketType : uint16_t { State = 1, Echo = 2 };

#pragma pack(push, 1)
struct Header {
    uint32_t magic;
    uint16_t version;
    uint16_t type;
};
struct StatePacket {
    Header h;
    uint32_t seq;
    uint32_t tsMs;
    float pos[3];
    float vel[3];
    float yaw;
    float pitch;
    uint32_t flags;
    uint32_t intents;
    uint16_t weapon;
    uint16_t pad;
};
struct EchoPacket {
    Header h;
    uint32_t ackSeq;
    uint32_t gtaTimeMs;
    float pos[3];
};
#pragma pack(pop)

static_assert(sizeof(StatePacket) == 60);
static_assert(sizeof(EchoPacket) == 28);

inline bool ParseState(const void* data, size_t len, StatePacket& out) {
    if (len != sizeof(StatePacket)) {
        return false;
    }
    std::memcpy(&out, data, sizeof(out));
    return out.h.magic == kMagic && out.h.version == kVersion
        && out.h.type == static_cast<uint16_t>(PacketType::State);
}

inline EchoPacket MakeEcho(uint32_t ackSeq, uint32_t gtaTimeMs, float x, float y, float z) {
    EchoPacket e{};
    e.h = {kMagic, kVersion, static_cast<uint16_t>(PacketType::Echo)};
    e.ackSeq = ackSeq;
    e.gtaTimeMs = gtaTimeMs;
    e.pos[0] = x;
```

```cpp
#pragma once
#include <algorithm>
#include <cmath>
#include <cstdint>

#include "protocol.h"

namespace cj {

struct Vec3 {
    float x = 0.0f;
    float y = 0.0f;
    float z = 0.0f;
};
inline Vec3 operator+(Vec3 a, Vec3 b) { return {a.x + b.x, a.y + b.y, a.z + b.z}; }
inline Vec3 operator-(Vec3 a, Vec3 b) { return {a.x - b.x, a.y - b.y, a.z - b.z}; }

// Minecraft displacement (x east, y up, z south) -> GTA (x east, y north, z up). Mirrors cjproto.mc_delta_to_gta.
inline Vec3 McDeltaToGta(Vec3 d) { return {d.x, -d.z, d.y}; }

// Decides where CJ must be for each Minecraft state. Pure logic: no game memory, no sockets.
class Puppet {
public:
    static constexpr uint32_t kTimeoutMs = 1000;
    static constexpr float kMaxCoord = 3.0e7f; // Minecraft world border
    static constexpr float kGtaLimitXY = 2900.0f;
    static constexpr float kGtaMinZ = -100.0f;
    static constexpr float kGtaMaxZ = 1500.0f;

    // Returns true and fills outTarget when the packet is accepted.
    bool Update(const StatePacket& s, uint32_t nowMs, Vec3 pedPos, Vec3& outTarget) {
        if (Expired(nowMs)) {
            Reset();
        }
        const Vec3 pos{s.pos[0], s.pos[1], s.pos[2]};
        const Vec3 vel{s.vel[0], s.vel[1], s.vel[2]};
        const float yaw = s.yaw;
        const float pitch = s.pitch;
        if (!IsSane(pos) || !IsSane(vel) || !std::isfinite(yaw) || !std::isfinite(pitch)) {
            return false;
        }
        if (haveSeq_ && static_cast<int32_t>(s.seq - lastSeq_) <= 0) {
            return false;
        }
        if (!anchored_) {
            anchored_ = true;
            mcAnchor_ = pos;
            gtaAnchor_ = pedPos;
        }
        haveSeq_ = true;
        lastSeq_ = s.seq;
        lastAcceptMs_ = nowMs;

        const Vec3 t = gtaAnchor_ + McDeltaToGta(pos - mcAnchor_);
        outTarget = {std::clamp(t.x, -kGtaLimitXY, kGtaLimitXY),
                     std::clamp(t.y, -kGtaLimitXY, kGtaLimitXY),
                     std::clamp(t.z, kGtaMinZ, kGtaMaxZ)};
        return true;
    }

    bool Expired(uint32_t nowMs) const { return anchored_ && nowMs - lastAcceptMs_ > kTimeoutMs; }
    void Reset() { *this = Puppet{}; }

private:
    static bool IsSane(Vec3 v) {
        return std::isfinite(v.x) && std::isfinite(v.y) && std::isfinite(v.z)
            && std::fabs(v.x) <= kMaxCoord && std::fabs(v.y) <= kMaxCoord && std::fabs(v.z) <= kMaxCoord;
    }

    bool anchored_ = false;
    Vec3 mcAnchor_{};
    Vec3 gtaAnchor_{};
    bool haveSeq_ = false;
    uint32_t lastSeq_ = 0;
    uint32_t lastAcceptMs_ = 0;
};

} // namespace cj
```

```cpp
#include "hook.h"

#include <windows.h>

#include <cstdint>
#include <cstring>

#include "bridge.h"
#include "log.h"
#include "ped_layout.h"

namespace cj {
namespace {

uintptr_t g_originalProcess = 0;

void __cdecl HookedProcess() {
    reinterpret_cast<void(__cdecl*)()>(g_originalProcess)();
    bridge::Tick();
}

// FindPlayerPed: mov eax,[esp+4]; test eax,eax; jge +7; movzx eax,byte [0xB7CD74]; imul eax,eax,0x190; mov eax,[eax+0xB7CD98]
bool AddressMapMatches() {
    static const uint8_t expected[] = {0x8B, 0x44, 0x24, 0x04, 0x85, 0xC0, 0x7D, 0x07, 0x0F, 0xB6, 0x05, 0x74, 0xCD,
                                       0xB7, 0x00, 0x69, 0xC0, 0x90, 0x01, 0x00, 0x00, 0x8B, 0x80, 0x98, 0xCD, 0xB7};
    return std::memcmp(reinterpret_cast<const void*>(gta::kFindPlayerPed), expected, sizeof(expected)) == 0;
}

} // namespace

bool InstallProcessHook() {
    if (!AddressMapMatches()) {
        Log("ADDRESS MISMATCH: FindPlayerPed @0x%X does not match gta-reversed; hook not installed", (unsigned)gta::kFindPlayerPed);
        return false;
    }
    auto* site = reinterpret_cast<uint8_t*>(gta::kProcessCallSite);
    if (site[0] != 0xE8) {
        Log("ADDRESS MISMATCH: 0x%X is not a call instruction (first byte 0x%02X)", (unsigned)gta::kProcessCallSite, site[0]);
        return false;
    }
    int32_t rel = 0;
    std::memcpy(&rel, site + 1, sizeof(rel));
    // Chain through whatever the call currently targets (another plugin may already have redirected it).
    g_originalProcess = gta::kProcessCallSite + 5 + rel;

    const int32_t newRel = static_cast<int32_t>(reinterpret_cast<uintptr_t>(&HookedProcess) - (gta::kProcessCallSite + 5));
    DWORD oldProtect = 0;
    if (!VirtualProtect(site, 5, PAGE_EXECUTE_READWRITE, &oldProtect)) {
        Log("VirtualProtect failed: %lu", GetLastError());
        return false;
    }
    std::memcpy(site + 1, &newRel, sizeof(newRel));
    VirtualProtect(site, 5, oldProtect, &oldProtect);
    FlushInstructionCache(GetCurrentProcess(), site, 5);
    Log("hook installed: call @0x%X now goes through the bridge (original target 0x%X)",
        (unsigned)gta::kProcessCallSite, (unsigned)g_originalProcess);
    return true;
}
```

```cpp
#include "bridge.h"

#include <winsock2.h>
#include <windows.h>

#include <cstdint>

#include "log.h"
#include "ped_layout.h"
#include "protocol.h"
#include "puppet.h"

namespace cj::bridge {
namespace {

constexpr uint16_t kGtaPort = 34560;  // we listen here
constexpr uint16_t kEchoPort = 34561; // Minecraft (or the fake client) listens here

SOCKET OpenSocket() {
    WSADATA wsa;
    if (WSAStartup(MAKEWORD(2, 2), &wsa) != 0) {
        Log("WSAStartup failed");
        return INVALID_SOCKET;
    }
    SOCKET s = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
    if (s == INVALID_SOCKET) {
        Log("socket() failed: %d", WSAGetLastError());
        return INVALID_SOCKET;
    }
    u_long nonBlocking = 1;
    ioctlsocket(s, FIONBIO, &nonBlocking);
    // Windows reports ICMP "port unreachable" from a previous sendto as a recv error; turn that off.
    BOOL noConnReset = FALSE;
    DWORD bytes = 0;
    WSAIoctl(s, 0x9800000C /* SIO_UDP_CONNRESET */, &noConnReset, sizeof(noConnReset), nullptr, 0, &bytes, nullptr, nullptr);

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
    addr.sin_port = htons(kGtaPort);
    if (bind(s, reinterpret_cast<sockaddr*>(&addr), sizeof(addr)) != 0) {
        Log("bind(127.0.0.1:%u) failed: %d", kGtaPort, WSAGetLastError());
        closesocket(s);
        return INVALID_SOCKET;
    }
    Log("listening on 127.0.0.1:%u, echoing to 127.0.0.1:%u", kGtaPort, kEchoPort);
    return s;
}

uint8_t* PlayerPed() {
    const uint8_t focus = *reinterpret_cast<volatile uint8_t*>(gta::kPlayerInFocus);
    if (focus >= gta::kMaxPlayers) {
        return nullptr;
    }
    auto* info = reinterpret_cast<uint8_t*>(gta::kPlayersArray) + focus * gta::kPlayerInfoSize;
    return *reinterpret_cast<uint8_t**>(info); // CPlayerInfo::m_pPed is the first member; null before a game is loaded
}

float* PedPosition(uint8_t* ped) {
    auto* matrix = *reinterpret_cast<uint8_t**>(ped + gta::kMatrixPtr);
    return reinterpret_cast<float*>(matrix ? matrix + gta::kMatrixPos : ped + gta::kPlacementPos);
}

void SetPhysicsEnabled(uint8_t* ped, bool enabled) {
    auto& entityFlags = *reinterpret_cast<uint8_t*>(ped + gta::kEntityFlags);
    auto& physicalFlags = *reinterpret_cast<uint32_t*>(ped + gta::kPhysicalFlags);
    if (enabled) {
        entityFlags |= 0x01;   // m_bUsesCollision
        physicalFlags |= 0x02; // bApplyGravity
    } else {
        entityFlags &= static_cast<uint8_t>(~0x01);
        physicalFlags &= ~0x02u;
        auto* speed = reinterpret_cast<float*>(ped + gta::kMoveSpeed);
        speed[0] = speed[1] = speed[2] = 0.0f;
    }
}

} // namespace

void Tick() {
    static Puppet puppet;
    static SOCKET sock = INVALID_SOCKET;
    static bool socketTried = false;
    static bool puppeted = false;
    static bool loggedNoPed = false;
    static uint32_t lastAck = 0;
    static uint32_t rejected = 0;
    static sockaddr_in echoTo = [] {
        sockaddr_in a{};
        a.sin_family = AF_INET;
        a.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
        a.sin_port = htons(kEchoPort);
        return a;
    }();

    if (!socketTried) {
        socketTried = true;
        sock = OpenSocket();
    }
    if (sock == INVALID_SOCKET) {
        return;
    }

    uint8_t* ped = PlayerPed();
    if (!ped && !loggedNoPed) {
        loggedNoPed = true;
        Log("no player ped yet; waiting");
    }
    if (ped) {
        loggedNoPed = false;
    }

    const uint32_t now = GetTickCount();
    float* pos = ped ? PedPosition(ped) : nullptr;
    Vec3 target{};
    bool haveTarget = false;

    // Drain every pending datagram so the socket never accumulates stale state.
    uint8_t buffer[256];
    for (;;) {
        const int n = recv(sock, reinterpret_cast<char*>(buffer), sizeof(buffer), 0);
        if (n == SOCKET_ERROR) {
            const int err = WSAGetLastError();
            if (err == WSAEMSGSIZE || err == WSAECONNRESET) {
                continue; // oversized garbage / stale ICMP error: skip it
            }
            break; // WSAEWOULDBLOCK: nothing left
        }
        if (!ped) {
            continue; // no CJ to move; discard
        }
        StatePacket state;
        if (!ParseState(buffer, static_cast<size_t>(n), state)) {
            if (++rejected % 100 == 1) {
                Log("rejected malformed datagram (%u so far, last size %d)", rejected, n);
            }
            continue;
        }
        Vec3 current{pos[0], pos[1], pos[2]};
        Vec3 t;
        if (puppet.Update(state, now, current, t)) {
            target = t;
            haveTarget = true;
            lastAck = state.seq;
        }
    }
    if (!ped) {
        return;
    }

    if (haveTarget) {
        if (!puppeted) {
            Log("puppet engaged at GTA (%.1f, %.1f, %.1f)", pos[0], pos[1], pos[2]);
            puppeted = true;
        }
        SetPhysicsEnabled(ped, false);
        pos[0] = target.x;
        pos[1] = target.y;
        pos[2] = target.z;
    } else if (puppeted && puppet.Expired(now)) {
        SetPhysicsEnabled(ped, true);
        puppet.Reset();
        puppeted = false;
        Log("no state for %u ms: puppet released, gravity and collision restored", Puppet::kTimeoutMs);
    }

    if (puppeted) {
        const EchoPacket echo = MakeEcho(lastAck, now, pos[0], pos[1], pos[2]);
        sendto(sock, reinterpret_cast<const char*>(&echo), sizeof(echo), 0,
               reinterpret_cast<const sockaddr*>(&echoTo), sizeof(echoTo));
    }
}

} // namespace cj::bridge
```

```python
#!/usr/bin/env python3
"""Stand-in for the Minecraft mod: drives the GTA bridge along a circle and checks the echoes."""
import argparse
import math
import random
import socket
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "protocol"))
import cjproto as p  # noqa: E402

GTA_ADDR = ("127.0.0.1", 34560)
ECHO_ADDR = ("127.0.0.1", 34561)
CENTER = (100.0, 64.0, 100.0)
RADIUS = 5.0
RATE_HZ = 60.0


def path(t):
    """Circle of radius 5 blocks in the XZ plane with a gentle vertical wobble; starts at path(0)."""
    return (CENTER[0] + RADIUS * math.cos(t), CENTER[1] + 0.5 * math.sin(2 * t), CENTER[2] + RADIUS * math.sin(t))


def drain_echoes(sock):
    echoes = []
    while True:
        try:
            data, _ = sock.recvfrom(256)
        except BlockingIOError:
            return echoes
        try:
            echoes.append(p.unpack_echo(data))
        except ValueError:
            pass


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--seconds", type=float, default=10.0)
    ap.add_argument("--tolerance", type=float, default=0.1, help="max distance in blocks between expected and real CJ")
    ap.add_argument("--garbage", action="store_true", help="interleave malformed, NaN and stale packets")
    args = ap.parse_args()

    rx = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    rx.bind(ECHO_ADDR)
    rx.setblocking(False)
    tx = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

    seq = 0

    def send_state(pos, sequence, **kw):
        state = p.State(seq=sequence, ts_ms=int(time.monotonic() * 1000) & 0xFFFFFFFF, pos=pos, vel=(0.0, 0.0, 0.0),
                        yaw=0.0, pitch=0.0, flags=p.FLAG_ON_GROUND, **kw)
        tx.sendto(p.pack_state(state), GTA_ADDR)

    # Phase 1: handshake. Every packet carries the same position, so whichever one the bridge anchors on is the same.
    p0 = path(0.0)
    anchor = None
    print("waiting for GTA to answer (load a game in GTA first) ...")
    deadline = time.monotonic() + 30.0
    while anchor is None and time.monotonic() < deadline:
        send_state(p0, seq)
        seq += 1
        time.sleep(0.05)
        echoes = drain_echoes(rx)
        if echoes:
            anchor = echoes[0].pos
    if anchor is None:
        print("FAIL: no echo from GTA within 30 s")
        return 2
    print(f"GTA answered; CJ anchored at GTA {anchor}")

    # Phase 2: drive the circle at 60 Hz and compare every echo with where CJ should be.
    history = {}
    checked = within = 0
    worst = 0.0
    start = time.monotonic()
    next_send = start
    while time.monotonic() - start < args.seconds:
        t = time.monotonic() - start
        pos = path(t)
        history[seq] = pos
        send_state(pos, seq)
        if args.garbage and seq % 20 == 0:
            tx.sendto(b"not a packet", GTA_ADDR)
            tx.sendto(p.pack_state(p.State(seq=seq + 1, ts_ms=0, pos=(float("nan"), 0.0, 0.0), vel=(0.0, 0.0, 0.0), yaw=0.0, pitch=0.0)), GTA_ADDR)
            tx.sendto(p.pack_state(p.State(seq=seq + 1, ts_ms=0, pos=(4.0e7, 0.0, 0.0), vel=(0.0, 0.0, 0.0), yaw=0.0, pitch=0.0)), GTA_ADDR)
            tx.sendto(p.pack_state(p.State(seq=seq - 5, ts_ms=0, pos=(0.0, 0.0, 0.0), vel=(0.0, 0.0, 0.0), yaw=0.0, pitch=0.0)), GTA_ADDR)
            tx.sendto(bytes(random.getrandbits(8) for _ in range(random.randint(1, 400))), GTA_ADDR)
        seq += 1
        for echo in drain_echoes(rx):
            sent = history.get(echo.ack_seq)
            if sent is None:
                continue
            delta = p.mc_delta_to_gta(tuple(a - b for a, b in zip(sent, p0)))
            expected = tuple(anchor[i] + delta[i] for i in range(3))
            err = math.dist(expected, echo.pos)
            worst = max(worst, err)
            checked += 1
            within += err <= args.tolerance
        next_send += 1.0 / RATE_HZ
        time.sleep(max(0.0, next_send - time.monotonic()))

    ratio = within / checked if checked else 0.0
    print(f"echoes checked: {checked}, within {args.tolerance} blocks: {within} ({ratio:.1%}), worst error: {worst:.3f}")
    ok = checked >= 50 and ratio >= 0.95
    print("PASS" if ok else "FAIL")
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main())
```

```java
package dev.cjmc;

import java.nio.ByteBuffer;
import java.nio.ByteOrder;

/** Wire format shared with the GTA bridge. See protocol/PROTOCOL.md. */
public final class Protocol {
    public static final int MAGIC = 0x434A4D43;
    public static final short VERSION = 1;
    public static final short TYPE_STATE = 1;
    public static final short TYPE_ECHO = 2;
    public static final int STATE_SIZE = 60;
    public static final int ECHO_SIZE = 28;

    public static final int FLAG_ON_GROUND = 1;
    public static final int FLAG_SPRINT = 2;
    public static final int FLAG_SNEAK = 4;
    public static final int FLAG_SWIM = 8;
    public static final int FLAG_CLIMBING = 16;

    public record State(int seq, int tsMs, float px, float py, float pz, float vx, float vy, float vz,
                        float yaw, float pitch, int flags, int intents, short weapon) {}

    public record Echo(int ackSeq, int gtaTimeMs, float x, float y, float z) {}

    private Protocol() {}

    public static byte[] encodeState(State s) {
        ByteBuffer b = ByteBuffer.allocate(STATE_SIZE).order(ByteOrder.LITTLE_ENDIAN);
        b.putInt(MAGIC).putShort(VERSION).putShort(TYPE_STATE);
        b.putInt(s.seq()).putInt(s.tsMs());
        b.putFloat(s.px()).putFloat(s.py()).putFloat(s.pz());
        b.putFloat(s.vx()).putFloat(s.vy()).putFloat(s.vz());
        b.putFloat(s.yaw()).putFloat(s.pitch());
        b.putInt(s.flags()).putInt(s.intents());
        b.putShort(s.weapon()).putShort((short) 0);
        return b.array();
    }

    /** Returns null when the datagram is not a valid echo. */
    public static Echo decodeEcho(byte[] data, int len) {
        if (len != ECHO_SIZE) {
            return null;
        }
        ByteBuffer b = ByteBuffer.wrap(data, 0, len).order(ByteOrder.LITTLE_ENDIAN);
        if (b.getInt() != MAGIC || b.getShort() != VERSION || b.getShort() != TYPE_ECHO) {
            return null;
        }
        return new Echo(b.getInt(), b.getInt(), b.getFloat(), b.getFloat(), b.getFloat());
    }
}
```

```java
package dev.cjmc;

import java.io.IOException;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.net.InetSocketAddress;
import java.util.Locale;
import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.rendering.v1.WorldRenderEvents;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.network.ClientPlayerEntity;
import net.minecraft.text.Text;
import net.minecraft.util.math.Vec3d;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class CjClient implements ClientModInitializer {
    public static final Logger LOG = LoggerFactory.getLogger("cj_fabric");

    private static final int GTA_PORT = 34560;
    private static final int ECHO_PORT = 34561;
    private static final long SEND_INTERVAL_NS = 1_000_000_000L / 60;
    private static final long HUD_INTERVAL_NS = 500_000_000L;
    private static final long ECHO_STALE_NS = 1_000_000_000L;

    private DatagramSocket socket;
    private InetAddress loopback;
    private int seq;
    private long nextSendNs;
    private long nextHudNs;
    private volatile Protocol.Echo lastEcho;
    private volatile long lastEchoNs;

    @Override
    public void onInitializeClient() {
        try {
            loopback = InetAddress.getByAddress(new byte[] {127, 0, 0, 1});
            socket = new DatagramSocket(new InetSocketAddress(loopback, ECHO_PORT));
        } catch (IOException e) {
            LOG.error("cannot open the bridge socket on 127.0.0.1:{} (is the fake client running?)", ECHO_PORT, e);
            return;
        }
        Thread receiver = new Thread(this::receiveLoop, "cj-bridge-rx");
        receiver.setDaemon(true);
        receiver.start();
        WorldRenderEvents.END.register(context -> onFrame());
        LOG.info("bridge ready: state -> 127.0.0.1:{}, echo <- 127.0.0.1:{}", GTA_PORT, ECHO_PORT);
    }

    private void receiveLoop() {
        byte[] buffer = new byte[256];
        DatagramPacket packet = new DatagramPacket(buffer, buffer.length);
        while (!socket.isClosed()) {
            try {
                packet.setLength(buffer.length);
                socket.receive(packet);
                Protocol.Echo echo = Protocol.decodeEcho(buffer, packet.getLength());
                if (echo != null) {
                    lastEcho = echo;
                    lastEchoNs = System.nanoTime();
                }
            } catch (IOException e) {
                if (socket.isClosed()) {
                    return;
                }
                LOG.warn("receive failed", e);
            }
        }
    }

    private void onFrame() {
        long now = System.nanoTime();
        if (now < nextSendNs) {
            return;
        }
        nextSendNs = now + SEND_INTERVAL_NS;

        MinecraftClient client = MinecraftClient.getInstance();
        ClientPlayerEntity player = client.player;
        if (player == null || client.world == null) {
            return;
        }

        float delta = client.getRenderTickCounter().getTickDelta(false);
        Vec3d pos = player.getLerpedPos(delta);
        Vec3d vel = player.getVelocity().multiply(20.0); // blocks/tick -> blocks/s
        int flags = 0;
        if (player.isOnGround()) flags |= Protocol.FLAG_ON_GROUND;
        if (player.isSprinting()) flags |= Protocol.FLAG_SPRINT;
        if (player.isSneaking()) flags |= Protocol.FLAG_SNEAK;
        if (player.isSwimming()) flags |= Protocol.FLAG_SWIM;
        if (player.isClimbing()) flags |= Protocol.FLAG_CLIMBING;

        Protocol.State state = new Protocol.State(seq++, (int) (now / 1_000_000L),
                (float) pos.x, (float) pos.y, (float) pos.z,
                (float) vel.x, (float) vel.y, (float) vel.z,
                player.getYaw(delta), player.getPitch(delta), flags, 0, (short) 0);
        byte[] data = Protocol.encodeState(state);
        try {
            socket.send(new DatagramPacket(data, data.length, loopback, GTA_PORT));
        } catch (IOException e) {
            LOG.warn("send failed", e);
        }

        if (now >= nextHudNs) {
            nextHudNs = now + HUD_INTERVAL_NS;
            showHud(player, now);
        }
    }

    private void showHud(ClientPlayerEntity player, long now) {
        Protocol.Echo echo = lastEcho;
        String text = (echo == null || now - lastEchoNs > ECHO_STALE_NS)
                ? "CJ bridge: GTA not responding"
                : String.format(Locale.ROOT, "CJ bridge: GTA (%.1f, %.1f, %.1f) ack=%d",
                        echo.x(), echo.y(), echo.z(), echo.ackSeq());
        player.sendMessage(Text.literal(text), true);
    }
}
```
